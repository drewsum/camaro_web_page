cmake_minimum_required(VERSION 3.10)
project(es-headless CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
add_compile_options("SHELL:-include cstring" "SHELL:-include cstdint" "SHELL:-include cmath" "SHELL:-include algorithm" -fdelayed-template-parsing -w -O2)
add_compile_definitions("__int64=long long" "__forceinline=inline __attribute__((always_inline))")
set(ES ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(Boost_USE_STATIC_LIBS OFF)
add_subdirectory(${ES}/dependencies/submodules/simple-2d-constraint-solver scs)
add_subdirectory(${ES}/dependencies/submodules/csv-io csvio)
add_subdirectory(${ES}/dependencies/submodules/piranha piranha)
set(CORE audio_buffer camshaft crankshaft combustion_chamber connecting_rod convolution_filter cylinder_bank cylinder_head delay_filter derivative_filter direct_throttle_linkage dynamometer engine exhaust_system feedback_comb_filter filter fuel function gas_system gaussian_filter governor ignition_module impulse_response intake jitter_filter leveling_filter low_pass_filter part piston piston_engine_simulator simulator standard_valvetrain starter_motor synthesizer throttle transmission utilities valvetrain vehicle vehicle_drag_constraint vtec_valvetrain)
set(SRCS)
foreach(f ${CORE})
  list(APPEND SRCS ${ES}/src/${f}.cpp)
endforeach()
add_library(es-core STATIC ${SRCS})
target_include_directories(es-core PUBLIC ${ES}/dependencies/submodules)
target_link_libraries(es-core simple-2d-constraint-solver csv-io)
add_library(es-script STATIC ${ES}/scripting/src/channel_types.cpp ${ES}/scripting/src/compiler.cpp ${ES}/scripting/src/engine_context.cpp ${ES}/scripting/src/language_rules.cpp)
target_compile_definitions(es-script PUBLIC ATG_ENGINE_SIM_PIRANHA_ENABLED)
target_include_directories(es-script PUBLIC ${ES}/dependencies/submodules)
target_link_libraries(es-script es-core csv-io piranha)
add_executable(es-render main.cpp)
target_link_libraries(es-render es-script es-core pthread)
target_link_options(es-render PRIVATE -Wl,--allow-multiple-definition)
